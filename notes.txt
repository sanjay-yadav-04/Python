q-find largest number from the input of the Usersfrom functools import reduce
l=[1,2,3,4,5]

n=int(input("Which  largest number You want:"))
if n <= len(l):
    for _ in range(n):
        x=reduce(lambda x,y:x if x>y else y,l)
        l.remove(x)
    print(f"The {n}th largest number is {x}")
else:
    print("please enter valid element")
